// REUSABLE POPUP WINDOW (DYNAMIC):

void popup(const std::vector<std::string>& lines) {
    int rows, cols;
    getmaxyx(stdscr, rows, cols);

    // find longest line
    int max_width = 0;
    for (const auto& line : lines)
        max_width = std::max(max_width, (int)line.length());

    // add padding
    int height = lines.size() + 4;
    int width  = max_width + 4;

    // center it
    int starty = (rows - height) / 2;
    int startx = (cols - width) / 2;

    WINDOW* win = newwin(height, width, starty, startx);
    box(win, 0, 0);

    // print lines centered inside the window
    for (int i = 0; i < (int)lines.size(); i++) {
        mvwprintw(win, i + 2, 2, "%s", lines[i].c_str());
    }

    mvwprintw(win, height - 2, 2, "Press any key...");
    wrefresh(win);

    wgetch(win);
    delwin(win);

    touchwin(stdscr);
    refresh();
}
